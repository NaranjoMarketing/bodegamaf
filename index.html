/** * API REST BODEGA MAF - VERSIÓN FINAL CON PEDIDOS EMPRESA Y GESTIÓN EN VIVO */ const SHEET_INVENTARIO = 'INVENTARIO_ACTUAL'; const SHEET_ENTRADAS = 'ENTRADAS'; const SHEET_SALIDAS = 'SALIDAS'; const SHEET_KARDEX = 'MOVIMIENTOS_KARDEX'; const SHEET_CONTACTOS = 'CONTACTOS'; const SHEET_CIERRES = 'CIERRES_DIARIOS'; const SHEET_PEDIDOS = 'PEDIDOS_EMPRESA'; const SHEET_AUTORIZADORES = 'AUTORIZADORES'; function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); try { return jsonResponse({ success: true, inventario: getSheetData(ss.getSheetByName(SHEET_INVENTARIO)), contactos: getSheetData(ss.getSheetByName(SHEET_CONTACTOS)), movimientos: getSheetData(ss.getSheetByName(SHEET_KARDEX)), cierres: getSheetData(ss.getSheetByName(SHEET_CIERRES)), pedidos: getSheetData(ss.getSheetByName(SHEET_PEDIDOS)), autorizadores: getSheetData(ss.getSheetByName(SHEET_AUTORIZADORES)) }); } catch (err) { return jsonResponse({ success: false, error: err.toString() }); } } function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const lock = LockService.getScriptLock(); if (!lock.tryLock(10000)) { return jsonResponse({ success: false, error: 'Servidor ocupado. Intenta de nuevo.' }); } try { const data = JSON.parse(e.postData.contents); const action = data.action; // --- 1. GUARDAR PEDIDO CORPORATIVO DESDE 'solicitud.html' --- if (action === 'guardarPedidoEmpresa') { const sheetPedidos = ss.getSheetByName(SHEET_PEDIDOS); if (sheetPedidos) { sheetPedidos.appendRow([ data.idPedido, data.fechaHora, data.empresa, data.reclama, data.autoriza, data.productos, data.destino || '', data.estado || 'Pendiente', data.idPedido ]); SpreadsheetApp.flush(); return jsonResponse({ success: true, message: 'Pedido registrado con éxito' }); } return jsonResponse({ success: false, error: 'Hoja PEDIDOS_EMPRESA no encontrada' }); } // --- 2. ACTUALIZAR ESTADO AL DESPACHAR EN BODEGA ('index.html') --- if (action === 'actualizarEstadoPedido') { const sheetPedidos = ss.getSheetByName(SHEET_PEDIDOS); const values = sheetPedidos.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < values.length; i++) { if (String(values[i][0]).trim() === String(data.idPedido).trim()) { rowIndex = i + 1; break; } } if (rowIndex > 0) { sheetPedidos.getRange(rowIndex, 8).setValue(data.nuevoEstado || 'Despachado'); SpreadsheetApp.flush(); return jsonResponse({ success: true, message: 'Estado actualizado' }); } return jsonResponse({ success: false, error: 'Pedido no encontrado' }); } // --- 3. GUARDAR CIERRE DIARIO --- if (action === 'guardarCierre') { const sheet = ss.getSheetByName(SHEET_CIERRES); if (sheet) { sheet.appendRow([ data.idCierre || ('CIE-' + Utilities.formatDate(new Date(), 'GMT-5', 'yyyyMMdd-HHmm')), data.fechaHora || Utilities.formatDate(new Date(), 'GMT-5', 'yyyy-MM-dd HH:mm:ss'), data.fechaJornada || Utilities.formatDate(new Date(), 'GMT-5', 'yyyy-MM-dd'), data.operador || 'Operador', Number(data.totalIngresos || 0), Number(data.totalSalidas || 0), Number(data.totalOrdenes || 0), Number(data.lineasActivas || 0), data.detalleMovimientos || '', data.estado || 'Cuadrado' ]); SpreadsheetApp.flush(); return jsonResponse({ success: true, message: 'Cierre diario guardado' }); } return jsonResponse({ success: false, error: 'Hoja CIERRES_DIARIOS no encontrada' }); } // --- 4. GUARDAR / EDITAR PRODUCTO EN EL CATÁLOGO --- if (action === 'guardarProducto') { const sheet = ss.getSheetByName(SHEET_INVENTARIO); const values = sheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < values.length; i++) { if (String(values[i][0]).trim() === String(data.sku).trim()) { rowIndex = i + 1; break; } } if (rowIndex > 0) { sheet.getRange(rowIndex, 2).setValue(data.nombre); sheet.getRange(rowIndex, 3).setValue(data.categoria); sheet.getRange(rowIndex, 4).setValue(data.unidad); if (data.stockMinimo !== undefined) { sheet.getRange(rowIndex, 9).setValue(Number(data.stockMinimo)); } SpreadsheetApp.flush(); return jsonResponse({ success: true, message: 'Producto modificado' }); } else { const newRow = sheet.getLastRow() + 1; const stockInicial = Number(data.stockInicial || 0); const stockMin = Number(data.stockMinimo || 10); const formulaEntradas = '=SUMIF(ENTRADAS!$C:$C, A' + newRow + ', ENTRADAS!$G:$G)'; const formulaSalidas = '=SUMIF(SALIDAS!$C:$C, A' + newRow + ', SALIDAS!$G:$G)'; const formulaActual = '=E' + newRow + ' + F' + newRow + ' - G' + newRow; const formulaEstado = '=IF(H' + newRow + '<=0, "Agotado", IF(H' + newRow + '<=I' + newRow + ', "Alerta Mínimo", "Disponible"))'; sheet.appendRow([ data.sku, data.nombre, data.categoria, data.unidad, stockInicial, formulaEntradas, formulaSalidas, formulaActual, stockMin, formulaEstado ]); SpreadsheetApp.flush(); return jsonResponse({ success: true, message: 'Producto creado' }); } } // --- 5. ELIMINAR PRODUCTO --- if (action === 'eliminarProducto') { const sheet = ss.getSheetByName(SHEET_INVENTARIO); const values = sheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < values.length; i++) { if (String(values[i][0]).trim() === String(data.sku).trim()) { rowIndex = i + 1; break; } } if (rowIndex > 0) { sheet.deleteRow(rowIndex); SpreadsheetApp.flush(); return jsonResponse({ success: true, message: 'Producto eliminado' }); } return jsonResponse({ success: false, error: 'Producto no encontrado' }); } // --- 6. REGISTRAR ENTRADA FLUVIAL --- if (action === 'registrarEntrada') { const sheetEntradas = ss.getSheetByName(SHEET_ENTRADAS); const sheetKardex = ss.getSheetByName(SHEET_KARDEX); const timestamp = new Date(); const fechaStr = Utilities.formatDate(timestamp, 'GMT-5', 'yyyy-MM-dd HH:mm'); data.items.forEach(item => { const idEntrada = data.remision || ('ING-' + Utilities.formatDate(timestamp, 'GMT-5', 'yyyyMMdd-HHmmss')); sheetEntradas.appendRow([ idEntrada, fechaStr, item.sku, item.nombre, item.unidad, data.lancha || 'Transporte Fluvial', Number(item.cantidad), 'Bodega Central', data.remision || idEntrada, data.recibidoPor ]); sheetKardex.appendRow([ fechaStr, idEntrada, 'ENTRADA', item.sku, item.nombre, item.unidad, Number(item.cantidad), data.lancha ? 'Entrega: ' + data.lancha : 'Ingreso directo', data.recibidoPor, 'Ingreso Bodega' ]); }); SpreadsheetApp.flush(); return jsonResponse({ success: true, message: 'Ingreso guardado' }); } // --- 7. REGISTRAR SALIDA DIRECTA --- if (action === 'registrarSalida') { const sheetSalidas = ss.getSheetByName(SHEET_SALIDAS); const sheetKardex = ss.getSheetByName(SHEET_KARDEX); const timestamp = new Date(); const fechaStr = Utilities.formatDate(timestamp, 'GMT-5', 'yyyy-MM-dd HH:mm'); const numOrden = 'DESP-' + Utilities.formatDate(timestamp, 'GMT-5', 'yyyyMMdd-HHmmss'); data.items.forEach(item => { const idSalida = 'SAL-' + Utilities.formatDate(timestamp, 'GMT-5', 'yyyyMMdd-HHmmss') + '-' + Math.floor(Math.random() * 100); sheetSalidas.appendRow([ idSalida, fechaStr, item.sku, item.nombre, item.unidad, numOrden, Number(item.cantidad), data.solicitante + (data.destino ? ' / ' + data.destino : ''), 'Retiro en Bodega', data.autorizadoPor, data.solicitante, 'Entregado' ]); sheetKardex.appendRow([ fechaStr, numOrden, 'SALIDA', item.sku, item.nombre, item.unidad, Number(item.cantidad), data.solicitante + (data.destino ? ' (' + data.destino + ')' : ''), data.autorizadoPor, 'Orden de Egreso' ]); }); SpreadsheetApp.flush(); return jsonResponse({ success: true, numOrden: numOrden, message: 'Orden registrada' }); } // --- 8. GUARDAR CONTACTO --- if (action === 'guardarContacto') { const sheet = ss.getSheetByName(SHEET_CONTACTOS); const idContacto = data.id || ('CONT-' + Math.floor(Math.random() * 9000 + 1000)); sheet.appendRow([ idContacto, data.nombre, data.rol, data.direccion || 'Inírida', data.telefono, data.observaciones || '' ]); SpreadsheetApp.flush(); return jsonResponse({ success: true, message: 'Contacto guardado' }); } // --- 9. ELIMINAR CONTACTO --- if (action === 'eliminarContacto') { const sheet = ss.getSheetByName(SHEET_CONTACTOS); const values = sheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < values.length; i++) { const idSheet = String(values[i][0]).trim(); const nombreSheet = String(values[i][1]).trim(); if ((data.id && idSheet === String(data.id).trim()) || (data.nombre && nombreSheet.toLowerCase() === String(data.nombre).trim().toLowerCase())) { rowIndex = i + 1; break; } } if (rowIndex > 0) { sheet.deleteRow(rowIndex); SpreadsheetApp.flush(); return jsonResponse({ success: true, message: 'Contacto eliminado' }); } return jsonResponse({ success: false, error: 'Contacto no encontrado' }); } return jsonResponse({ success: false, error: 'Acción no válida' }); } catch (err) { return jsonResponse({ success: false, error: err.toString() }); } finally { lock.releaseLock(); } } function getSheetData(sheet) { if (!sheet) return []; const values = sheet.getDataRange().getValues(); if (values.length <= 1) return []; const headers = values[0]; const rows = []; for (let i = 1; i < values.length; i++) { const rowObj = {}; for (let j = 0; j < headers.length; j++) { rowObj[headers[j]] = values[i][j]; } rows.push(rowObj); } return rows; } function jsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
